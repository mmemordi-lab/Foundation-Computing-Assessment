

// ASSESSMENT 2 GRADE B PROGRAM 
#include <stdio.h>

int main()
{
    // Declare variables
    char movieType;
    char ticketType;
    float ticketPrice=0.0;
    int NumOfTickets;
    float TotalBeforeDiscount;
    char SnackDecision;
    float Discount=0.0;
    float SnackCost=0.0;
    float FinalTotal;
    
    //ASK FOR USER INPUT

 printf("\n==== SOL ASSESSMENT 2 ====\n");
     printf("===== CINEMA TICKET BOOKING SYSTEM =====\n\n");

     printf("   ==== GRADE B ====\n");

    
    printf("Enter Movie Type (A = Action, C = Comedy, H = Horror): ");
    scanf(" %c", &movieType);

    printf("Enter Ticket Type (S = Standard, P = Premium): ");
    scanf(" %c", &ticketType);

    printf("Enter Number of Tickets: ");
    scanf("%d", &NumOfTickets);
    
 // Validate number of tickets
    if(NumOfTickets <=0)
    {
        printf ("\n Invalid Ticket Number");
        return 0;
    }
    
    // Determine Ticket Price 

    switch(movieType)
    {
        case 'A':
        case 'a':

            if(ticketType=='S' || ticketType=='s')
                ticketPrice = 10.00;
            else if(ticketType=='P' || ticketType=='p')
                ticketPrice = 15.00;
            else
            {
                printf("\nInvalid Ticket Type!\n");
                return 0;
            }

            break;

        case 'C':
        case 'c':

            if(ticketType=='S' || ticketType=='s')
                ticketPrice = 8.00;
            else if(ticketType=='P' || ticketType=='p')
                ticketPrice = 13.00;
            else
            {
                printf("\nInvalid Ticket Type!\n");
                return 0;
            }

            break;

        case 'H':
        case 'h':

            if(ticketType=='S' || ticketType=='s')
                ticketPrice = 9.00;
            else if(ticketType=='P' || ticketType=='p')
                ticketPrice = 14.00;
            else
            {
                printf("\nInvalid Ticket Type!\n");
                return 0;
            }

            break;

        default:

            printf("\nInvalid Movie Type!\n");
            return 0;
    }

    TotalBeforeDiscount = NumOfTickets * ticketPrice;

    // Group Discount 

    if(NumOfTickets > 6)
    {
      Discount = TotalBeforeDiscount * 0.10;
    }

    // Snack Combo 

    printf("Need a Snack Combo? (y/n): ");
    scanf(" %c", &SnackDecision);

    if(SnackDecision=='y' || SnackDecision=='Y')
    {
        SnackCost = NumOfTickets * 5.50;
    }
    else if(SnackDecision=='n' || SnackDecision=='N')
    {
        SnackCost = 0;
    }
    else
    {
        printf("\nInvalid Snack Decision!\n");
        return 0;
    }

 FinalTotal = TotalBeforeDiscount - Discount + SnackCost;

    // FINAL DESIGN OUTPUT 
    printf("\n *****BOOKING RESULT ******\n");

    
    switch(movieType)
    {
        case 'A':
        case 'a':
            printf("Movie Type            : Action\n");
            break;

        case 'C':
        case 'c':
            printf("Movie Type            : Comedy\n");
            break;

        case 'H':
        case 'h':
            printf("Movie Type            : Horror\n");
            break;
    }

    switch(ticketType)
    {
        case 'S':
        case 's':
            printf("Ticket Type           : Standard Seat\n");
            break;

        case 'P':
        case 'p':
            printf("Ticket Type           : Premium Seat\n");
            break;
    }

    printf("Number of Tickets     : %d\n", NumOfTickets);

    printf("Total Before Discount : £%.2f\n", TotalBeforeDiscount);

    printf("10% Discount          : £%.2f\n", Discount);

    printf("Snack Combo Cost      : £%.2f\n", SnackCost);

    printf("Final Total           : £%.2f\n", FinalTotal);

    return 0;
}
